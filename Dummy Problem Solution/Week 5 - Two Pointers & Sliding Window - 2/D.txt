#include<bits/stdc++.h>
using namespace std;
int t,n,x,y;
int main(){
	scanf("%d",&t);
	while (t--){
		scanf("%d%d%d",&n,&x,&y);
		vector<int> a,b;
		for (int i=1;i<=n;i++){
			int v;
			scanf("%d",&v);
			if (i>x&&i<=y) b.push_back(v);
			else a.push_back(v);
		}
		rotate(b.begin(),min_element(b.begin(),b.end()),b.end());
		auto it=a.begin();
		while (it!=a.end()&&*it<b[0]) ++it;
		a.insert(it,b.begin(),b.end());
		for (int i=0;i<n;i++) printf("%d ",a[i]);
		printf("\n");
	}
}
