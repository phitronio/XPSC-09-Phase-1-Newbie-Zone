#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int t,n,a,b,x,s;
ll w[300009];
int main(){
	ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
	cin>>t;
	while(t--){
		cin>>n,s=0;
		for(int i=1;i<=n;++i) cin>>w[i];
		a=(w[1]+w[n])/(n-1);
		cout<<(a-w[1]+w[2])/2<<" ";
		for(int i=2;i<n;++i) cout<<(w[i-1]+w[i+1]-2*w[i])/2<<" ";
		cout<<(a-w[n]+w[n-1])/2<<"\n";
	}
	return 0;
}
